cmake_minimum_required(VERSION 3.21)
project(WorkshopBankReference LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Bank reference requires MSVC x86.")
endif()
set(BANK_PAYLOAD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.local/generated" CACHE PATH "Generated selector header directory")
if(NOT EXISTS "${BANK_PAYLOAD_DIR}/bank_payload.hpp")
    message(FATAL_ERROR "Generate the native payload from the built H5U with scripts/build-player.py first.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(XKIT_GRAPHICS_FILE "" CACHE FILEPATH "Trusted xkit d3d9.dll build artifact shipped with this release")
if(XKIT_GRAPHICS_FILE)
    if(NOT EXISTS "${XKIT_GRAPHICS_FILE}")
        message(FATAL_ERROR "Build the pinned xkit graphics DLL before configuring this release.")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${XKIT_GRAPHICS_FILE}")
    file(SHA256 "${XKIT_GRAPHICS_FILE}" xkit_graphics_digest)
endif()
add_executable(workshop_bank_reference src/bank_launcher.cpp)
target_compile_definitions(workshop_bank_reference PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(workshop_bank_reference PRIVATE devkit/native "${BANK_PAYLOAD_DIR}")
target_link_libraries(workshop_bank_reference PRIVATE bcrypt comdlg32 user32)
install(TARGETS workshop_bank_reference RUNTIME DESTINATION tools
        COMPONENT Diagnostics EXCLUDE_FROM_ALL)
add_library(workshop_bank_plugin SHARED src/bank_launcher.cpp)
set_target_properties(workshop_bank_plugin PROPERTIES OUTPUT_NAME "WorkshopBankReference")
target_compile_definitions(workshop_bank_plugin PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
if(XKIT_GRAPHICS_FILE)
    target_compile_definitions(workshop_bank_plugin PRIVATE XKIT_GRAPHICS_SHA256="${xkit_graphics_digest}")
endif()
target_include_directories(workshop_bank_plugin PRIVATE devkit/native "${BANK_PAYLOAD_DIR}")
target_link_libraries(workshop_bank_plugin PRIVATE bcrypt comdlg32 user32)
install(TARGETS workshop_bank_plugin RUNTIME DESTINATION bin/Heroes5Mods)
include(CTest)
if(BUILD_TESTING)
    add_executable(bank_selector_test tests/bank_selector_test.cpp)
    target_compile_definitions(bank_selector_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(bank_selector_test PRIVATE devkit/native "${BANK_PAYLOAD_DIR}")
    target_link_libraries(bank_selector_test PRIVATE bcrypt comdlg32 user32)
    add_test(NAME native_bank_selector COMMAND bank_selector_test)
endif()
